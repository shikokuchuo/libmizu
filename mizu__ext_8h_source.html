990MIZU_EXT_INLINE int mizu_ext_mizl_ent(const void *base, size_t size,
991 int64_t i, mizu_mizl_entry *e) {
992 const unsigned char *dir = (const unsigned char *) base +
993 MIZU_HEADER_SIZE + 32 * (size_t) i;
994 memcpy(&e->data_offset, dir, 8);
995 memcpy(&e->data_size, dir + 8, 8);
996 memcpy(&e->sexptype, dir + 16, 4);
997 memcpy(&e->attrs_size, dir + 20, 4);
998 memcpy(&e->length, dir + 24, 8);
999 if (e->data_offset < 0 || (e->data_offset & 63) != 0 ||
1000 e->data_size < 0 ||
1001 (uint64_t) e->data_offset > (uint64_t) size ||
1002 (uint64_t) e->data_size >
1003 (uint64_t) size - (uint64_t) e->data_offset ||
1004 e->attrs_size < 0)
1005 return -1;
1006 if (!mizu_ext_mizl_tag_ok(e->sexptype)) return -1;
1007 const int32_t tag = e->sexptype & ~(int32_t) MIZU_MIZL_S4;
1008 if (tag == 33) { /* remote leaf: the span is the identifier */
1009 if ((e->sexptype & (int32_t) MIZU_MIZL_S4) != 0 || e->length < 0 ||
1010 e->data_size < 1 || e->data_size > 255)
1011 return -1;
1012 return 0;
1013 }
1014 if ((int64_t) e->attrs_size > e->data_size) return -1;
1015 const int64_t body = e->data_size - (int64_t) e->attrs_size;
1016 const size_t elt = mizu_type_elt_size(tag);
1017 if (elt != 0) {
1018 if (e->length < 0 || e->length > body / (int64_t) elt) return -1;
1019 } else if (tag == MIZU_TYPE_STR) {
1020 if (e->length < 0 || e->length > ((int64_t) 1 << 50) ||
1021 mizu_mizs_geometry(e->length).data > body)
1022 return -1;
1023 } else if (tag == MIZU_TYPE_VEC) {
1024 if (e->length < 0) return -1;
1025 }
1026 return 0;
1027}

1126 (uint64_t) voff > (uint64_t) size ||

1182MIZU_EXT_INLINE uint64_t mizu_ext_na_build_impl(int type, uint8_t *bitmap,
1183 const void *src, uint64_t n,

1187 const uint64_t bit = bit_off + i;
1188 if (mizu_ext_na_at(type, src, i)) {
1189 bitmap[bit >> 3] &= (uint8_t) ~(1u << (bit & 7));
1190 nulls++;
1191 } else {
1192 bitmap[bit >> 3] |= (uint8_t) (1u << (bit & 7));
1193 }
1194 }
1195 return nulls;

1200 const uint8_t *bitmap,

1207 if (((bitmap[i >> 3] >> (i & 7)) & 1) == 0) {
1208 mizu_ext_na_store(type, dst, i);
1209 nulls++;
1210 }
1211 }
1212 return nulls;
