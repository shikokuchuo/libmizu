1091 e->attrs_size < 0)
1092 return -1;
1093 if (!mizu_ext_mizl_tag_ok(e->sexptype)) return -1;
1094 const int32_t tag = e->sexptype & ~(int32_t) MIZU_MIZL_S4;
1095 if (tag == 33) { /* remote leaf: the span is the identifier */
1096 if ((e->sexptype & (int32_t) MIZU_MIZL_S4) != 0 || e->length < 0 ||
1097 e->data_size < 1 || e->data_size > 255)
1098 return -1;
1099 return 0;
1100 }
1101 if ((int64_t) e->attrs_size > e->data_size) return -1;
1102 const int64_t body = e->data_size - (int64_t) e->attrs_size;
1103 const size_t elt = mizu_type_elt_size(tag);
1104 if (elt != 0) {
1105 if (e->length < 0 || e->length > body / (int64_t) elt) return -1;
1106 } else if (tag == MIZU_TYPE_STR) {
1107 if (e->length < 0 || e->length > ((int64_t) 1 << 50) ||
1108 mizu_mizs_geometry(e->length).data > body)
1109 return -1;

1111 if (e->length < 0) return -1;
1112 }

1177MIZU_EXT_INLINE int mizu_ext_mizl_check_impl(const void *base, size_t size,
1178 int64_t *n, int64_t *attrs_off,
1179 int64_t *attrs_size,
1180 int64_t valid[2]) {
1181 if (size < MIZU_HEADER_SIZE) return -1;
1182 uint32_t magic;
1183 int32_t n32;
1184 int64_t aoff, asz, voff, vcount;
1185 memcpy(&magic, base, 4);
1186 if (magic != MIZU_MAGIC_LIST) return -1;
1187 memcpy(&n32, (const unsigned char *) base + 4, 4);
1188 memcpy(&aoff, (const unsigned char *) base + 8, 8);
1189 memcpy(&asz, (const unsigned char *) base + 16, 8);
1190 memcpy(&voff, (const unsigned char *) base + MIZU_HDR_VALID_OFF, 8);
1191 memcpy(&vcount, (const unsigned char *) base + MIZU_HDR_VALID_COUNT, 8);
1192 if (n32 < 0 ||
1193 (uint64_t) (uint32_t) n32 >
1194 ((uint64_t) size - (uint64_t) MIZU_HEADER_SIZE) / 32)
1195 return -1;
1196 if (aoff < 0 || asz < 0 ||
1197 (uint64_t) aoff > (uint64_t) size ||
1198 (uint64_t) asz > (uint64_t) size - (uint64_t) aoff)
1199 return -1;
1200 if (!mizu_ext_flags_known(base)) return -1;
1201 uint64_t sum = 0;
1202 for (int64_t i = 0; i < (int64_t) n32; i++) {
1203 mizu_mizl_entry e;
1204 if (mizu_ext_mizl_ent(base, size, i, &e) != 0) return -1;
1205 const int32_t tag = e.sexptype & ~(int32_t) MIZU_MIZL_S4;
1206 /* vcount is an int64, so the bound needs the sum only up to
1207 INT64_MAX: saturate. Overlapping crafted leaves in a huge region
1208 can otherwise wrap the accumulation and over-reject. */
1209 if (mizu_type_elt_size(tag) != 0 && sum < (uint64_t) INT64_MAX) {
1210 const uint64_t len = (uint64_t) e.length;
1211 sum = len > (uint64_t) INT64_MAX - sum ? (uint64_t) INT64_MAX
1212 : sum + len;
1213 }
1214 }
1215 if (voff == 0) {
1216 if (vcount != 0 && vcount != -1) return -1;
1217 } else {
1218 const uint64_t bytes = (uint64_t) (uint32_t) n32 * 16;
1219 if (vcount < 0 || (uint64_t) vcount > sum || (voff & 63) != 0 ||
1220 (uint64_t) voff > (uint64_t) size ||
1221 bytes > (uint64_t) size - (uint64_t) voff)
1222 return -1;
1223 }
1224 *n = n32;
1225 *attrs_off = aoff;
1226 *attrs_size = asz;
1227 valid[0] = voff;
1228 valid[1] = vcount;
1229 return 0;
1230}

1232MIZU_EXT_INLINE int mizu_ext_mizl_elem_impl(const void *base, size_t size,
1233 int64_t i,
1234 mizu_mizl_entry *elem) {
1235 if (size < MIZU_HEADER_SIZE) return -1;
1236 uint32_t magic;
1237 int32_t n32;
1238 int64_t voff, vcount;
1239 memcpy(&magic, base, 4);
1240 if (magic != MIZU_MAGIC_LIST) return -1;
1241 memcpy(&n32, (const unsigned char *) base + 4, 4);
1242 if (n32 < 0 || i < 0 || i >= (int64_t) n32 ||
1243 (uint64_t) (uint32_t) n32 >
1244 ((uint64_t) size - (uint64_t) MIZU_HEADER_SIZE) / 32)
1245 return -1;
1246 if (!mizu_ext_flags_known(base)) return -1;
1247 if (mizu_ext_mizl_ent(base, size, i, elem) != 0) return -1;
1248 memcpy(&voff, (const unsigned char *) base + MIZU_HDR_VALID_OFF, 8);
1249 memcpy(&vcount, (const unsigned char *) base + MIZU_HDR_VALID_COUNT, 8);
1250 if (voff == 0) {
1251 if (vcount != 0 && vcount != -1) return -1;
1252 elem->valid[0] = 0;
1253 elem->valid[1] = vcount;
1254 return 0;
1255 }
1256 if ((voff & 63) != 0) return -1;
1257 const uint64_t bytes = (uint64_t) (uint32_t) n32 * 16;
1258 if ((uint64_t) voff > (uint64_t) size ||
1259 bytes > (uint64_t) size - (uint64_t) voff)
1260 return -1;
1261 const unsigned char *tab = (const unsigned char *) base + voff;
1262 int64_t loff, lcount;
1263 memcpy(&loff, tab + 16 * (size_t) i, 8);
1264 memcpy(&lcount, tab + 16 * (size_t) i + 8, 8);
1265 /* a remote leaf (tag 33) carries a {0,0} / {0,-1} claim alone — a
1266 bitmap offset is region-local and cannot describe a remote column.
1267 STR / VEC / serialized leaves carry no table bitmap either: the
1268 string block, the nested header, and the blob each speak for
1269 themselves, and the writers stamp those rows {0,0} */
1270 const int32_t ltag = elem->sexptype & ~(int32_t) MIZU_MIZL_S4;
1271 if ((ltag == 33 || mizu_type_elt_size(ltag) == 0) && loff != 0)
1272 return -1;
1273 if (!mizu_ext_valid_ok(loff, lcount, (uint64_t) elem->length, size))
1274 return -1;
1275 elem->valid[0] = loff;
1276 elem->valid[1] = lcount;
1277 return 0;
1278}

1647MIZU_EXT_INLINE int mizu_ext_ix_utf8_valid_impl(const void *s, size_t n) {
1650 const unsigned char *b = (const unsigned char *) s;
1651 size_t i = 0;
1652 while (i < n) {
1653 const unsigned char c = b[i];
1654 if (c < 0x80) {
1655 i++;
1656 } else if (c < 0xC2 || c >= 0xF5) { /* stray cont., overlong, or
1657 past U+10FFFF */
1658 return 0;
1659 } else if (c < 0xE0) { /* 2 bytes */
1660 if (i + 2 > n || (b[i + 1] & 0xC0) != 0x80) return 0;
1661 i += 2;
1662 } else if (c < 0xF0) { /* 3 bytes */
1663 if (i + 3 > n || (b[i + 1] & 0xC0) != 0x80 ||
1664 (b[i + 2] & 0xC0) != 0x80) return 0;
1665 if (c == 0xE0 && b[i + 1] < 0xA0) return 0; /* overlong */
1666 if (c == 0xED && b[i + 1] >= 0xA0) return 0; /* surrogate */
1667 i += 3;
1668 } else { /* 4 bytes: 0xF0..0xF4 */
1669 if (i + 4 > n || (b[i + 1] & 0xC0) != 0x80 ||
1670 (b[i + 2] & 0xC0) != 0x80 || (b[i + 3] & 0xC0) != 0x80)
1671 return 0;

1675 }
1676 }

1920MIZU_EXT_INLINE size_t mizu_ext_ix_write_err_impl(
1921 unsigned char *dst, uint32_t inline_max,
1922 const void *type, size_t type_n,
1923 const void *msg, size_t msg_n,
1924 const void *detail, size_t detail_n,
1925 int has_index, uint64_t index) {
1931 const size_t budget = inline_max;
1932 const size_t avail =
1933 budget > MIZU_IX_ERR_OVERHEAD ? budget - MIZU_IX_ERR_OVERHEAD : 0;
1934 size_t cap =
1935 avail < MIZU_IX_ERR_TYPE_SHARE ? avail : MIZU_IX_ERR_TYPE_SHARE;
1936 const size_t tn = mizu_ext_ix_err_floor_impl(type, type_n, cap);
1937 cap = budget / 2;
1938 if (cap > avail - tn) cap = avail - tn;
1939 const size_t mn = mizu_ext_ix_err_floor_impl(msg, msg_n, cap);
1940 const size_t dn =
1941 mizu_ext_ix_err_floor_impl(detail, detail_n, avail - tn - mn);
1942 size_t n = mizu_ext_ix_put_header_impl(dst);
1943 n += mizu_ext_ix_put_err_impl(dst != NULL ? dst + n : NULL, has_index,
1944 index, type, (uint32_t) tn, msg,
1945 (uint32_t) mn, detail, (uint32_t) dn);
1946 return n;

2179MIZU_EXT_INLINE int mizu_ext_arrow_str_at_impl(const ArrowArray *arr,
2180 int64_t k, int form,
2181 const uint8_t **ptr,
2182 int32_t *len) {
2183 if (!mizu_ext_arrow_valid_impl(arr, k)) return 0;
2184 int64_t i = arr->offset + k;
2185 if (form == 1) {
2186 const uint8_t *offs = (const uint8_t *) arr->buffers[1];
2187 int32_t o0, o1;
2188 memcpy(&o0, offs + 4 * i, 4);
2189 memcpy(&o1, offs + 4 * (i + 1), 4);
2190 if (o0 < 0 || o1 < o0) return -1;
2191 *ptr = (const uint8_t *) arr->buffers[2] + o0;
2192 *len = o1 - o0;
2193 } else if (form == 2) {
2194 const uint8_t *offs = (const uint8_t *) arr->buffers[1];
2195 int64_t o0, o1;
2196 memcpy(&o0, offs + 8 * i, 8);
2197 memcpy(&o1, offs + 8 * (i + 1), 8);
2198 if (o0 < 0 || o1 < o0 || o1 - o0 > INT32_MAX)
2199 return -1;
2200 *ptr = (const uint8_t *) arr->buffers[2] + o0;
2201 *len = (int32_t) (o1 - o0);
2202 } else {
2203 const uint8_t *vw = (const uint8_t *) arr->buffers[1] + i * 16;
2204 int32_t l;
2205 memcpy(&l, vw, 4);
2206 if (l < 0) return -1;
2207 if (l <= 12) {
2208 *ptr = vw + 4;
2209 } else {
2210 int32_t bi, off;
2211 memcpy(&bi, vw + 8, 4);
2212 memcpy(&off, vw + 12, 4);
2213 /* The variadic data buffers are slots [2, n_buffers): the C Data
2214 spec's trailing slot is the int64 sizes array, but a producer
2215 that skips it (a nanoarrow nested or standalone export) puts a
2216 data buffer there. The reference tells the layouts apart: a
2217 spec-layout view never names the trailing slot, so one that
2218 does proves the no-sizes layout, its span the producer's word;
2219 any other span the sizes array bounds by the declared size. */
2220 if (bi < 0 || off < 0 || 2 + (int64_t) bi >= arr->n_buffers)
2221 return -1;
2222 if (2 + bi < arr->n_buffers - 1) {
2223 const uint8_t *sizes =
2224 (const uint8_t *) arr->buffers[arr->n_buffers - 1];
2225 if (sizes != NULL) {
2226 int64_t sz;
2227 memcpy(&sz, sizes + 8 * (ptrdiff_t) bi, 8);
2228 if ((int64_t) off + l > sz) return -1;
2229 }
2230 }
2231 *ptr = (const uint8_t *) arr->buffers[2 + bi] + off;
2232 }
2233 *len = l;
2234 }
2235 return 1;
