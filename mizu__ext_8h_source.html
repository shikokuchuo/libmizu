1082 const int32_t tag = e->sexptype & ~(int32_t) MIZU_MIZL_S4;

1084 if ((e->sexptype & (int32_t) MIZU_MIZL_S4) != 0 || e->length < 0 ||
1085 e->data_size < 1 || e->data_size > 255)
1086 return -1;
1087 return 0;
1088 }
1089 if ((int64_t) e->attrs_size > e->data_size) return -1;
1090 const int64_t body = e->data_size - (int64_t) e->attrs_size;
1091 const size_t elt = mizu_type_elt_size(tag);
1092 if (elt != 0) {
1093 if (e->length < 0 || e->length > body / (int64_t) elt) return -1;

1095 if (e->length < 0 || e->length > ((int64_t) 1 << 50) ||

1099 if (e->length < 0) return -1;

1165MIZU_EXT_INLINE int mizu_ext_mizl_check_impl(const void *base, size_t size,
1166 int64_t *n, int64_t *attrs_off,
1167 int64_t *attrs_size,
1168 int64_t valid[2]) {
1169 if (size < MIZU_HEADER_SIZE) return -1;
1170 uint32_t magic;
1171 int32_t n32;
1172 int64_t aoff, asz, voff, vcount;
1173 memcpy(&magic, base, 4);
1174 if (magic != MIZU_MAGIC_LIST) return -1;
1175 memcpy(&n32, (const unsigned char *) base + 4, 4);
1176 memcpy(&aoff, (const unsigned char *) base + 8, 8);
1177 memcpy(&asz, (const unsigned char *) base + 16, 8);
1178 memcpy(&voff, (const unsigned char *) base + MIZU_HDR_VALID_OFF, 8);
1179 memcpy(&vcount, (const unsigned char *) base + MIZU_HDR_VALID_COUNT, 8);
1180 if (n32 < 0 ||
1181 (uint64_t) (uint32_t) n32 >
1182 ((uint64_t) size - (uint64_t) MIZU_HEADER_SIZE) / 32)
1183 return -1;
1184 if (aoff < 0 || asz < 0 ||
1185 (uint64_t) aoff > (uint64_t) size ||
1186 (uint64_t) asz > (uint64_t) size - (uint64_t) aoff)
1187 return -1;
1188 if (!mizu_ext_flags_known(base)) return -1;
1189 uint64_t sum = 0;
1190 for (int64_t i = 0; i < (int64_t) n32; i++) {
1191 mizu_mizl_entry e;
1192 if (mizu_ext_mizl_ent(base, size, i, &e) != 0) return -1;
1193 const int32_t tag = e.sexptype & ~(int32_t) MIZU_MIZL_S4;
1194 /* vcount is an int64, so the bound needs the sum only up to
1195 INT64_MAX: saturate. Overlapping crafted leaves in a huge region
1196 can otherwise wrap the accumulation and over-reject. */
1197 if (mizu_type_elt_size(tag) != 0 && sum < (uint64_t) INT64_MAX) {
1198 const uint64_t len = (uint64_t) e.length;
1199 sum = len > (uint64_t) INT64_MAX - sum ? (uint64_t) INT64_MAX
1200 : sum + len;
1201 }
1202 }
1203 if (voff == 0) {
1204 if (vcount != 0 && vcount != -1) return -1;
1205 } else {
1206 const uint64_t bytes = (uint64_t) (uint32_t) n32 * 16;
1207 if (vcount < 0 || (uint64_t) vcount > sum || (voff & 63) != 0 ||
1208 (uint64_t) voff > (uint64_t) size ||
1209 bytes > (uint64_t) size - (uint64_t) voff)
1210 return -1;
1211 }
1212 *n = n32;
1213 *attrs_off = aoff;
1214 *attrs_size = asz;
1215 valid[0] = voff;
1216 valid[1] = vcount;
1217 return 0;
1218}

1220MIZU_EXT_INLINE int mizu_ext_mizl_elem_impl(const void *base, size_t size,
1221 int64_t i,
1222 mizu_mizl_entry *elem) {
1223 if (size < MIZU_HEADER_SIZE) return -1;
1224 uint32_t magic;
1225 int32_t n32;
1226 int64_t voff, vcount;
1227 memcpy(&magic, base, 4);
1228 if (magic != MIZU_MAGIC_LIST) return -1;
1229 memcpy(&n32, (const unsigned char *) base + 4, 4);
1230 if (n32 < 0 || i < 0 || i >= (int64_t) n32 ||
1231 (uint64_t) (uint32_t) n32 >
1232 ((uint64_t) size - (uint64_t) MIZU_HEADER_SIZE) / 32)
1233 return -1;
1234 if (!mizu_ext_flags_known(base)) return -1;
1235 if (mizu_ext_mizl_ent(base, size, i, elem) != 0) return -1;
1236 memcpy(&voff, (const unsigned char *) base + MIZU_HDR_VALID_OFF, 8);
1237 memcpy(&vcount, (const unsigned char *) base + MIZU_HDR_VALID_COUNT, 8);
1238 if (voff == 0) {
1239 if (vcount != 0 && vcount != -1) return -1;
1240 elem->valid[0] = 0;
1241 elem->valid[1] = vcount;
1242 return 0;
1243 }
1244 if ((voff & 63) != 0) return -1;
1245 const uint64_t bytes = (uint64_t) (uint32_t) n32 * 16;
1246 if ((uint64_t) voff > (uint64_t) size ||
1247 bytes > (uint64_t) size - (uint64_t) voff)
1248 return -1;
1249 const unsigned char *tab = (const unsigned char *) base + voff;
1250 int64_t loff, lcount;
1251 memcpy(&loff, tab + 16 * (size_t) i, 8);
1252 memcpy(&lcount, tab + 16 * (size_t) i + 8, 8);
1253 /* a remote leaf (tag 33) carries a {0,0} / {0,-1} claim alone — a
1254 bitmap offset is region-local and cannot describe a remote column.
1255 STR / VEC / serialized leaves carry no table bitmap either: the
1256 string block, the nested header, and the blob each speak for
1257 themselves, and the writers stamp those rows {0,0} */
1258 const int32_t ltag = elem->sexptype & ~(int32_t) MIZU_MIZL_S4;
1259 if ((ltag == 33 || mizu_type_elt_size(ltag) == 0) && loff != 0)
1260 return -1;
1261 if (!mizu_ext_valid_ok(loff, lcount, (uint64_t) elem->length, size))
1262 return -1;
1263 elem->valid[0] = loff;
1264 elem->valid[1] = lcount;
1265 return 0;
1266}

1634MIZU_EXT_INLINE int mizu_ext_ix_utf8_valid_impl(const void *s, size_t n) {
1637 const unsigned char *b = (const unsigned char *) s;
1638 size_t i = 0;
1639 while (i < n) {
1640 const unsigned char c = b[i];
1641 if (c < 0x80) {
1642 i++;
1643 } else if (c < 0xC2 || c >= 0xF5) { /* stray cont., overlong, or
1644 past U+10FFFF */
1645 return 0;
1646 } else if (c < 0xE0) { /* 2 bytes */
1647 if (i + 2 > n || (b[i + 1] & 0xC0) != 0x80) return 0;
1648 i += 2;
1649 } else if (c < 0xF0) { /* 3 bytes */
1650 if (i + 3 > n || (b[i + 1] & 0xC0) != 0x80 ||
1651 (b[i + 2] & 0xC0) != 0x80) return 0;
1652 if (c == 0xE0 && b[i + 1] < 0xA0) return 0; /* overlong */
1653 if (c == 0xED && b[i + 1] >= 0xA0) return 0; /* surrogate */
1654 i += 3;
1655 } else { /* 4 bytes: 0xF0..0xF4 */
1656 if (i + 4 > n || (b[i + 1] & 0xC0) != 0x80 ||
1657 (b[i + 2] & 0xC0) != 0x80 || (b[i + 3] & 0xC0) != 0x80)
1658 return 0;
1659 if (c == 0xF0 && b[i + 1] < 0x90) return 0; /* overlong */
1660 if (c == 0xF4 && b[i + 1] > 0x8F) return 0; /* > U+10FFFF */

1908 unsigned char *dst, uint32_t inline_max,

1911 const void *detail, size_t detail_n,

1922 avail < MIZU_IX_ERR_TYPE_SHARE ? avail : MIZU_IX_ERR_TYPE_SHARE;
1923 const size_t tn = mizu_ext_ix_err_floor_impl(type, type_n, cap);
1924 cap = budget / 2;

1928 mizu_ext_ix_err_floor_impl(detail, detail_n, avail - tn - mn);

1930 n += mizu_ext_ix_put_err_impl(dst != NULL ? dst + n : NULL, has_index,
1931 index, type, (uint32_t) tn, msg,
1932 (uint32_t) mn, detail, (uint32_t) dn);

2171 int64_t i = arr->offset + k;

2173 const uint8_t *offs = (const uint8_t *) arr->buffers[1];
2174 int32_t o0, o1;
2175 memcpy(&o0, offs + 4 * i, 4);
2176 memcpy(&o1, offs + 4 * (i + 1), 4);
2177 if (o0 < 0 || o1 < o0) return -1;
2178 *ptr = (const uint8_t *) arr->buffers[2] + o0;
2179 *len = o1 - o0;
2180 } else if (form == 2) {
2181 const uint8_t *offs = (const uint8_t *) arr->buffers[1];
2182 int64_t o0, o1;
2183 memcpy(&o0, offs + 8 * i, 8);
2184 memcpy(&o1, offs + 8 * (i + 1), 8);
2185 if (o0 < 0 || o1 < o0 || o1 - o0 > INT32_MAX)
2186 return -1;

2191 int32_t l;

2201 spec's trailing slot is the int64 sizes array, but a producer
2202 that skips it (a nanoarrow nested or standalone export) puts a

2204 spec-layout view never names the trailing slot, so one that
