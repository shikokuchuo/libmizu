783typedef struct mizu_task_s {
784 uint64_t word;
785} mizu_task;
786
787#define MIZU_TASK_SEQ_BITS 40
788#define MIZU_TASK_SEQ_MAX ((UINT64_C(1) << MIZU_TASK_SEQ_BITS) - 1)
789
790static inline uint64_t mizu_task_seq(const mizu_task *t) {
791 return t->word & MIZU_TASK_SEQ_MAX;
792}
793static inline uint32_t mizu_task_rs_index(const mizu_task *t) {
794 return (uint32_t) (t->word >> MIZU_TASK_SEQ_BITS);
795}
796static inline mizu_task mizu_task_make(uint64_t seq, uint32_t rs_index) {
797 mizu_task t;
798 t.word = (seq & MIZU_TASK_SEQ_MAX) |
799 ((uint64_t) rs_index << MIZU_TASK_SEQ_BITS);
800 return t;
801}
