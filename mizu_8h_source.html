694#define MIZU_TASK_SEQ_BITS 40
695#define MIZU_TASK_SEQ_MAX ((UINT64_C(1) << MIZU_TASK_SEQ_BITS) - 1)
696
697static inline uint64_t mizu_task_seq(const mizu_task *t) {
698 return t->word & MIZU_TASK_SEQ_MAX;
699}
700static inline uint32_t mizu_task_rs_index(const mizu_task *t) {
701 return (uint32_t) (t->word >> MIZU_TASK_SEQ_BITS);
702}
703static inline mizu_task mizu_task_make(uint64_t seq, uint32_t rs_index) {
704 mizu_task t;
705 t.word = (seq & MIZU_TASK_SEQ_MAX) |
706 ((uint64_t) rs_index << MIZU_TASK_SEQ_BITS);
707 return t;
708}
