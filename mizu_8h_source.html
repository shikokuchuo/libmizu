702#define MIZU_TASK_SEQ_BITS 40
703#define MIZU_TASK_SEQ_MAX ((UINT64_C(1) << MIZU_TASK_SEQ_BITS) - 1)
704
705static inline uint64_t mizu_task_seq(const mizu_task *t) {
706 return t->word & MIZU_TASK_SEQ_MAX;
707}
708static inline uint32_t mizu_task_rs_index(const mizu_task *t) {
709 return (uint32_t) (t->word >> MIZU_TASK_SEQ_BITS);
710}
711static inline mizu_task mizu_task_make(uint64_t seq, uint32_t rs_index) {
712 mizu_task t;
713 t.word = (seq & MIZU_TASK_SEQ_MAX) |
714 ((uint64_t) rs_index << MIZU_TASK_SEQ_BITS);
715 return t;
716}
