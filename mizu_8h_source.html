769#define MIZU_TASK_SEQ_BITS 40
770#define MIZU_TASK_SEQ_MAX ((UINT64_C(1) << MIZU_TASK_SEQ_BITS) - 1)
771
772static inline uint64_t mizu_task_seq(const mizu_task *t) {
773 return t->word & MIZU_TASK_SEQ_MAX;
774}
775static inline uint32_t mizu_task_rs_index(const mizu_task *t) {
776 return (uint32_t) (t->word >> MIZU_TASK_SEQ_BITS);
777}
778static inline mizu_task mizu_task_make(uint64_t seq, uint32_t rs_index) {
779 mizu_task t;
780 t.word = (seq & MIZU_TASK_SEQ_MAX) |
781 ((uint64_t) rs_index << MIZU_TASK_SEQ_BITS);
782 return t;
783}
