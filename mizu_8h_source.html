847#define MIZU_TASK_SEQ_BITS 40
848#define MIZU_TASK_SEQ_MAX ((UINT64_C(1) << MIZU_TASK_SEQ_BITS) - 1)
849
850static inline uint64_t mizu_task_seq(const mizu_task *t) {
851 return t->word & MIZU_TASK_SEQ_MAX;
852}
853static inline uint32_t mizu_task_rs_index(const mizu_task *t) {
854 return (uint32_t) (t->word >> MIZU_TASK_SEQ_BITS);
855}
856static inline mizu_task mizu_task_make(uint64_t seq, uint32_t rs_index) {
857 mizu_task t;
858 t.word = (seq & MIZU_TASK_SEQ_MAX) |
859 ((uint64_t) rs_index << MIZU_TASK_SEQ_BITS);
860 return t;
861}
